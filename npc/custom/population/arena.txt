// Shared arena manager. Each player reserves one room and chooses every job.
-	script	Population Arena	-1,{
	mes "[Population Arena]";
	.@action = select("Start a match:Stop my match:Cancel");
	if (.@action == 3) close;
	if (.@action == 2) {
		callsub S_Stop;
		mes "Your match has ended.";
		close;
	}
	if (@pop_arena_map$ != "") { mes "Stop your current match first."; close; }
	.@choice = select("Arena 1:Arena 2:Arena 3:Arena 4:Arena 5:Cancel");
	if (.@choice == 6) close;
	.@map$ = "pvp_y_1-" + .@choice;
	.@room = mapname2id(.@map$);
	if (.@room <= 0) { mes "This arena is unavailable."; close; }
	if ($@pop_arena_owner[.@room]) { mes "This arena is occupied. Choose another room."; close; }
	.@size = select("1v1:3v3:5v5:Cancel");
	if (.@size == 4) close;
	.@enemies = .@size * 2 - 1;
	explode(.@ids$, population_arena_jobs(), ",");
	.@nj = getarraysize(.@ids$);
	if (.@nj == 1 && .@ids$[0] == "") .@nj = 0;
	.@menu$ = "Random Mix";
	for (.@i = 0; .@i < .@nj; ++.@i) .@menu$ += ":" + jobname(atoi(.@ids$[.@i]));
	.@menu$ += ":Cancel";
	// Collect choices before reserving the room or creating any shells.
	for (.@i = 0; .@i < 2 * .@enemies - 1; ++.@i) {
		mes .@i < .@enemies ? "Choose opponent " + (.@i + 1) + "." : "Choose ally " + (.@i - .@enemies + 1) + ".";
		next;
		.@pick = select(.@menu$);
		if (.@pick == .@nj + 2) close;
		.@jobs[.@i] = .@pick == 1 ? 0 : atoi(.@ids$[.@pick - 2]);
	}
	if ($@pop_arena_owner[.@room]) { mes "Another player reserved this room."; close; }
	$@pop_arena_owner[.@room] = getcharid(0);
	@pop_arena_map$ = .@map$;
	@pop_arena_had_pvp = getmapflag(.@map$, mf_pvp);
	getmapxy @pop_arena_return$, @pop_arena_x, @pop_arena_y, BL_PC;
	setmapflag .@map$, mf_pvp;
	.@spawned = 0;
	for (.@i = 0; .@i < 2 * .@enemies - 1; ++.@i)
		.@spawned += population_arena_start(.@map$, 1, 0, 0, .@jobs[.@i], .@i < .@enemies ? 1 : 2);
	if (.@spawned < 2 * .@enemies - 1) {
		callsub S_Stop;
		mes "The match could not be created. Check population capacity and job profiles.";
		close;
	}
	close2;
	warp .@map$, 0, 0;
	end;

S_Stop:
	if (@pop_arena_map$ == "") return;
	.@room = mapname2id(@pop_arena_map$);
	if ($@pop_arena_owner[.@room] == getcharid(0)) {
		population_arena_stop(@pop_arena_map$);
		if (!@pop_arena_had_pvp) removemapflag @pop_arena_map$, mf_pvp;
		$@pop_arena_owner[.@room] = 0;
	}
	@pop_arena_map$ = "";
	if (!@pop_arena_logging_out && @pop_arena_return$ != "") warp @pop_arena_return$, @pop_arena_x, @pop_arena_y;
	return;

OnPCLogoutEvent:
	@pop_arena_logging_out = 1;
	callsub S_Stop;
	end;
}
